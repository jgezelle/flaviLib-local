2. Parse cmsearch tblout files

read_tblout <- function(path, cm_label) {
  lines <- readLines(path)
  lines <- lines[!startsWith(lines, "#") & nchar(trimws(lines)) > 0]
  if (length(lines) == 0) return(NULL)
  do.call(rbind, lapply(lines, function(l) {
    p <- strsplit(trimws(l), "\\s+")[[1]]
    data.frame(
      fasta_id  = p[1],
      E_value   = as.numeric(p[16]),
      score     = as.numeric(p[15]),
      inc       = p[17],
      cm_class  = cm_label,
      stringsAsFactors = FALSE
    )
  }))
}

3. CM annotation of input sequences

Each sequence assigned to CM class with best (lowest) E-value. Unmatched = Unannotated.

cm_filt <- cm_all %>%
  { if (SIG_ONLY) filter(., inc == "!") else . }

# Per accession: best hit across all CMs
cm_best <- cm_filt %>%
  group_by(accession) %>%
  slice_min(E_value, n = 1, with_ties = FALSE) %>%
  ungroup() %>%
  select(accession, cm_class, E_value)

# Join taxonomy with CM hits; exclude Tombusviridae
acc_input <- tax %>%
  filter(family != "Tombusviridae") %>%
  left_join(cm_best, by = "accession") %>%
  mutate(
    cm_class = ifelse(is.na(cm_class), "Unannotated", cm_class),
    cm_class = factor(cm_class, levels = CM_LEVELS),
    genus    = ifelse(is.na(genus) | genus == "", "Unannotated", genus)
  )

# Genus order by total count descending
genus_order <- acc_input %>%
  count(genus, sort = TRUE) %>%
  pull(genus)

acc_input <- acc_input %>%
  mutate(genus = factor(genus, levels = genus_order))
